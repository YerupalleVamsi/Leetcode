#include <vector>
#include <string>
#include <algorithm>
using namespace std;

class Solution {
private:
    bool isValidCode(string s) {
        if (s.empty()) {
            return false;
        }
        for (char c : s) {
            if (c >= 'a' && c <= 'z') continue;
            if (c >= 'A' && c <= 'Z') continue;
            if (c >= '0' && c <= '9') continue;
            if (c == '_') continue;
            return false;
        }
        return true;
    }
public:
    vector<string> validateCoupons(vector<string>& code, vector<string>& businessLine, vector<bool>& isActive) {
        vector<string> validBL = {"electronics", "grocery", "pharmacy", "restaurant"};
        vector<pair<int, string>> validCoupons;
        int n = code.size();
        for (int i = 0; i < n; ++i) {
            if (!isActive[i]) continue;
            
            string bl = businessLine[i];
            int p = -1;
            for (int j = 0; j < 4; ++j) {
                if (bl == validBL[j]) {
                    p = j;
                    break;
                }
            }
            if (p == -1) continue;
            
            if (!isValidCode(code[i])) continue;
            
            validCoupons.push_back(make_pair(p, code[i]));
        }
        
        sort(validCoupons.begin(), validCoupons.end());
        
        vector<string> result;
        for (auto& pr : validCoupons) {
            result.push_back(pr.second);
        }
        return result;
    }
};
