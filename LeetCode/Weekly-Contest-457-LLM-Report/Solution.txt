class Solution {
public:
    vector<string> validateCoupons(vector<string>& code, vector<string>& businessLine, vector<bool>& isActive) {
        unordered_map<string, int> priorityMap = {
            {"electronics", 0},
            {"grocery", 1},
            {"pharmacy", 2},
            {"restaurant", 3}
        };

        vector<pair<int, string>> validOnes;
        int n = code.size();

        for (int i = 0; i < n; i++) {
            if (!isActive[i]) 
                continue;

            string const& bl = businessLine[i];
            if (priorityMap.find(bl) == priorityMap.end())
                continue;

            string const& c = code[i];
            if (c.empty())
                continue;

            bool validCode = true;
            for (char ch : c) {
                if (ch >= 'a' && ch <= 'z') continue;
                if (ch >= 'A' && ch <= 'Z') continue;
                if (ch >= '0' && ch <= '9') continue;
                if (ch == '_') continue;
                validCode = false;
                break;
            }

            if (!validCode)
                continue;

            validOnes.push_back({priorityMap[bl], c});
        }

        sort(validOnes.begin(), validOnes.end(), [](const pair<int,string>& a, const pair<int,string>& b) {
            if (a.first != b.first)
                return a.first < b.first;
            return a.second < b.second;
        });

        vector<string> result;
        for (auto& p : validOnes) {
            result.push_back(p.second);
        }

        return result;
    }
};
